function iniciarApp() { actualizarFechaYHoraDigital(); calcularFaseLunar(); let gpsRespondio = false; if (navigator.geolocation) { // Intento principal exigiendo alta precisión y cero caché navigator.geolocation.getCurrentPosition( (position) => { gpsRespondio = true; procesarUbicacionGPS(position); }, (error) => { console.warn("GPS inicial demoró, buscando respaldo de Android..."); intentarRecuperarDesdeAndroid(); }, { timeout: 12000, maximumAge: 0, enableHighAccuracy: true } ); // Segundo intento de reescritura rápida por si el chip GPS tardó en despertar del reposo navigator.geolocation.getCurrentPosition( (position) => { if (position.coords.accuracy < 150) { // Si la precisión es confiable gpsRespondio = true; procesarUbicacionGPS(position); } }, null, { timeout: 15000, maximumAge: 0, enableHighAccuracy: true } ); } // Margen de seguridad por si el GPS no responde setTimeout(() => { if (!gpsRespondio) { intentarRecuperarDesdeAndroid(); } }, 6000); } function procesarUbicacionGPS(position) { currentLat = position.coords.latitude; currentLon = position.coords.longitude; obtenerNombreUbicacion(currentLat, currentLon); inicializarMapaArgentina(currentLat, currentLon); dispararCargaDeDatos(); } function intentarRecuperarDesdeAndroid() { if (window.Android && typeof window.Android.obtenerLatitudGuardada === "function") { currentLat = window.Android.obtenerLatitudGuardada(); currentLon = window.Android.obtenerLongitudGuardada(); const nombreAndroid = window.Android.obtenerNombreGuardado(); const elUbi = document.getElementById("ubicacionTexto"); if (elUbi) elUbi.innerText = nombreAndroid; inicializarMapaArgentina(currentLat, currentLon); dispararCargaDeDatos(); } else { currentLat = -32.92; currentLon = -68.84; document.getElementById("ubicacionTexto").innerText = "Godoy Cruz, Mendoza"; inicializarMapaArgentina(currentLat, currentLon); dispararCargaDeDatos(); } }
